  "[" EXP_NUMBER_TO(FROM) [ ":" EXP_NUMBER_TO(LENGTH) "]" ]                      -> "[" EXP_NUMBER_TO EXP_NUMBER_TO1 "]>>Command.Range?from;length=-1;"

  "[" EXP_NUMBER_TO(TO) "]"                                                      -> "["  EXP_NUMBER_TO "]>>Command.Range?from;len=0;"

a[1:] - substring from 1 to end
a[1:5] - substring from 1 to 1+5
a[-1:5] - substring from end i.e. 5 last
a[:5]   - substring 5 first
